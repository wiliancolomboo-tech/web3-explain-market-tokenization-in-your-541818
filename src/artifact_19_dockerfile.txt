# Multi-stage production build for Sovereign-RWA Node Orchestrator
FROM node:20-slim AS builder

WORKDIR /usr/src/app

COPY client/package*.json ./
COPY client/tsconfig.json ./
RUN npm ci

COPY client/src ./src
RUN npm run build

FROM node:20-slim AS runner

WORKDIR /usr/src/app
ENV NODE_ENV=production

COPY client/package*.json ./
RUN npm ci --only=production

COPY --from=builder /usr/src/app/dist ./dist

# Healthcheck for uptime monitor
HEALTHCHECK --interval=30s --timeout=5s --start-period=5s --retries=3 \
  CMD node -e "require('http').get('http://localhost:8080/healthz', (r) => {if (r.statusCode !== 200) process.exit(1)});"

USER node
EXPOSE 8080

CMD ["node", "dist/cli.js", "listen", "--mint", "MINT_PUBKEY_HERE", "--symbol", "US-CRE-001", "--email", "compliance@institution.com"]
